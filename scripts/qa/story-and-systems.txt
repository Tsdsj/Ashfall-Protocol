async (page) => {

 const aim=async(point)=>{await page.evaluate(point=>{const p=window.__game.sim.state.player,dx=point.x-p.position.x,dz=point.z-p.position.z;p.yaw=Math.atan2(dx,dz);p.pitch=Math.atan2(p.position.y+1.68-point.y,Math.hypot(dx,dz));},point);await page.waitForTimeout(120);};

 const door=async(id)=>{const p=await poi(id);await resume();await move(p.x,p.z-p.depth/2-2.8);await aim({x:p.x,y:p.y+1.3,z:p.z-p.depth/2});if(!await page.evaluate(id=>window.__game.sim.state.doors[id],id))await page.keyboard.press('e');return p;};
 const crate=async(id)=>{const p=await door(id);await move(p.x-p.width/2+2.8,p.z+p.depth/2-3.8);const c=await page.evaluate(id=>window.__game.sim.state.containers[id+':0'],id);await aim(c.position);await page.keyboard.press('e');check((await inspect()).screen==='inventory','crate not opened '+id);return c;};
 const take=async(cid,itemId)=>{const uid=await page.evaluate(({cid,itemId})=>window.__game.sim.state.containers[cid].inventory.items.find(i=>i.id===itemId)?.uid,{cid,itemId});check(!!uid,'missing required loot '+itemId);await page.locator(`[data-action="select-item"][data-source="${cid}"][data-uid="${uid}"]`).dblclick();};
 try{
  await page.reload();await page.getByRole('button',{name:'▸ 继续生存 ↗'}).waitFor();await page.bringToFront();await page.getByRole('button',{name:'▸ 继续生存 ↗'}).click();await page.locator('[data-action="resume"]').waitFor();await resume();await page.evaluate(()=>{window.__game.command('god');window.__game.command('time 12');window.__game.command('weather clear');});
  await door('lab-0');check(!await page.evaluate(()=>window.__game.sim.state.doors['lab-0']),'lab opened without keycard');results.push({name:'研究站访问卡门禁',pass:true});
  const fort=await crate('fort-0');await take(fort.id,'keycard');check(await count('keycard')===1,'keycard not transferred');results.push({name:'军事区搜刮访问卡',pass:true});
  const lab=await door('lab-0');check(await page.evaluate(()=>window.__game.sim.state.doors['lab-0']===true),'keycard did not open lab');await move(lab.x+3,lab.z-.8);await aim({x:lab.x+3,y:lab.y+.96,z:lab.z+1.7});await page.waitForFunction(()=>window.__game.renderer.target()?.id==='lab');await page.keyboard.press('e');await page.waitForTimeout(180);check((await inspect()).screen==='journal','research record not read');check(await count('protocol')===1,'protocol missing');results.push({name:'研究站档案与背包连接',pass:true});
  const industry=await crate('industry-0');for(const item of ['radio','electronics','battery'])await take(industry.id,item);results.push({name:'工业区广播维修物资',pass:true});
  const station=await door('broadcast');await move(station.x,station.z-.7);await aim({x:station.x,y:station.y+1.5,z:station.z+1.6});await page.waitForFunction(()=>window.__game.renderer.target()?.type==='radio');await page.keyboard.press('e');await page.waitForTimeout(100);check((await inspect()).flags.includes('broadcast'),'broadcast did not complete');check(await count('radio')===0,'radio not consumed');results.push({name:'修复广播、消耗材料、发送协议',pass:true});
  const extraction=await poi('extraction');await move(extraction.x+6,extraction.z-2.2);await aim({x:extraction.x+6,y:extraction.y+.55,z:extraction.z});await page.keyboard.press('e');check((await inspect()).screen==='end','extraction did not complete');await page.screenshot({path:'output/playwright/ending-verified.png',scale:'css'});await page.getByRole('button',{name:'留在灰谷，继续生存'}).click();results.push({name:'主线结局与继续游玩',pass:true});
  const clinic=await door('pine-1');await move(clinic.x-2,clinic.z-4.2);await aim({x:clinic.x-2,y:clinic.y+1.15,z:clinic.z-2});await page.keyboard.press('e');check((await inspect()).screen==='trade','NPC trade not opened');check(await count('rawmeat')>=3,'test save lacks meat');const meat=await count('rawmeat');await page.locator('[data-action="trade"][data-index="4"]').click();check(await count('rawmeat')===meat-3,'barter input not consumed');check(await count('fuel')>=2,'barter output not granted');results.push({name:'幸存者交易与物资守恒',pass:true});
  await resume();const vehicle=await page.evaluate(()=>window.__game.sim.state.vehicles.find(v=>v.id==='starter-pickup'));await move(vehicle.position.x+3,vehicle.position.z);await aim({...vehicle.position,y:vehicle.position.y+1});await page.keyboard.press('e');check((await inspect()).screen==='vehicle','vehicle UI not opened');await page.locator('[data-action="service-vehicle"][data-kind="fuel"]').click();check(await count('fuel')===1,'vehicle fuel did not consume resource');await page.locator('[data-action="enter-vehicle"]').click();const before=await inspect();await page.keyboard.down('w');await page.waitForTimeout(1800);await page.keyboard.up('w');await page.keyboard.down('Space');await page.waitForTimeout(900);await page.keyboard.up('Space');const after=await inspect();check(Math.hypot(after.position.x-before.position.x,after.position.z-before.position.z)>1,'vehicle did not move');await page.keyboard.press('e');check((await inspect()).vehicle===null,'could not exit stopped vehicle');results.push({name:'车辆加油、驾驶、制动与下车',pass:true,distance:Math.hypot(after.position.x-before.position.x,after.position.z-before.position.z)});
  await page.keyboard.press('m');await page.locator('#world-map').click({position:{x:200,y:200}});check(await page.evaluate(()=>!!window.__game.sim.state.waypoint),'map marker did not persist in world state');await page.screenshot({path:'output/playwright/map-verified.png',scale:'css'});results.push({name:'地图探索、导航标记',pass:true});
  await resume();await move(-120,-72);await page.evaluate(()=>{for(const id of ['kit_generator','kit_light','kit_storage','kit_planter','kit_wire'])window.__game.command('give '+id);window.__game.command('give fuel 3');window.__game.command('give wood 5');});
  for(const [kind,x,z]of [['generator',-123,-75],['light',-119,-75],['storage',-114,-75],['planter',-109,-75]]){
   await resume();await move(x,z);await page.evaluate(()=>window.__game.sim.state.player.pitch=.65);await page.waitForTimeout(100);await page.keyboard.press('b');await page.locator(`[data-action="choose-building"][data-id="${kind}"]`).click();await page.waitForTimeout(120);const preview=await page.evaluate(()=>window.__game.renderer.placement());check(preview.valid,'invalid placement '+kind+': '+preview.reason);await page.keyboard.press('e');await page.waitForTimeout(80);await page.keyboard.press('Escape');
  }
  const interactStructure=async(kind)=>{const b=await page.evaluate(kind=>window.__game.sim.state.structures.find(b=>b.id.startsWith('build-')&&b.kind===kind),kind);await resume();await move(b.position.x,b.position.z-2.2);await aim({...b.position,y:b.position.y+(kind==='light'?2.6:kind==='planter'?.28:.55)});await page.waitForFunction(id=>window.__game.renderer.target()?.id===id,b.id);await page.keyboard.press('e');await page.waitForTimeout(150);check((await inspect()).screen==='structure','structure not interactive '+kind);return b;};
  const gen=await interactStructure('generator');await page.locator('[data-action="structure-toggle"]').click();check(await page.evaluate(id=>window.__game.sim.state.structures.find(b=>b.id===id).active,gen.id),'generator did not start');const light=await interactStructure('light');await page.locator('[data-action="structure-toggle"]').click();check(await page.evaluate(pos=>window.__game.sim.building.powered(pos),light.position),'power network did not reach light');results.push({name:'发电机、消耗燃料、范围供电与照明',pass:true});
  const box=await interactStructure('storage');await page.locator('[data-action="open-structure-storage"]').click();const wood=await page.evaluate(()=>window.__game.sim.state.player.inventory.items.find(i=>i.id==='wood').uid);await page.locator(`[data-action="select-item"][data-source="player"][data-uid="${wood}"]`).click({modifiers:['Shift']});check(await page.evaluate(id=>window.__game.sim.state.structures.find(b=>b.id===id).inventory.items.some(i=>i.id==='wood'),box.id),'storage transfer failed');results.push({name:'基地储物箱与背包转移',pass:true});
  const planter=await interactStructure('planter');await page.locator('[data-action="structure-toggle"]').click();check(await page.evaluate(id=>window.__game.sim.state.structures.find(b=>b.id===id).plantedAt>0,planter.id),'seed was not planted');results.push({name:'种植与种子消耗',pass:true});
  await page.keyboard.press('Escape');await page.keyboard.press('Escape');await page.getByRole('button',{name:'保存生存记录',exact:true}).click();await page.waitForTimeout(250);return{pass:true,setup:'Story travel and aim use developer positioning, God mode prevents unrelated combat interrupts. Extra building kits were supplied only for the independent power/storage/planting checks. All gates, transfers, terminal actions, driving, barter, placement and device controls use UI input.',results};
 }catch(error){const state=await page.evaluate(()=>window.__game?.inspect());await page.screenshot({path:'output/playwright/systems-failure.png',scale:'css'});await page.keyboard.press('Escape');return{pass:false,error:String(error),state,toasts:await page.locator('.toasts').textContent(),target:await page.evaluate(()=>window.__game?.renderer.target()),results};}
}

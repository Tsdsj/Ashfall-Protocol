async(page)=>{
const results=[],errors=[];page.on('pageerror',e=>errors.push(String(e)));
await page.evaluate(async()=>{const g=window.__game,saves=await g.game.saves.list();const base=saves.find(s=>s.name==='QA 剧情场景联调'&&s.state.narrative.sequenceFlags.includes('finale-ready'));if(!base)throw Error('No completed mainline fixture');await g.game.load(base.id);g.sim.god=true;});
await page.locator('[data-action="resume"]').click();await page.waitForTimeout(200);
async function step(id){
if(await page.evaluate(async id=>{const{NARRATIVE_INTERACTIONS}=await import('/src/narrative/quests.ts');const d=NARRATIVE_INTERACTIONS.find(d=>d.id===id);return d.questId&&(window.__game.sim.state.narrative.quests[d.questId]?.stage??0)>d.stage;},id)){results.push({id,alreadyCompleted:true});return;}
await page.evaluate(async id=>{const g=window.__game;g.game.input.unlock();g.game.input.fallback=true;g.ui.show('play');g.ui.setLocked(true);const it=g.sim.narrative.interactions().find(i=>i.id==='narrative:'+id);if(!it)throw Error('missing '+id);let view=null;for(const length of [2.05,1.6,1.2])for(const yaw of [0,Math.PI/2,Math.PI,-Math.PI/2]){const p={x:it.position.x-Math.sin(yaw)*length,y:it.position.y,z:it.position.z-Math.cos(yaw)*length};if(!view&&!g.sim.collision.blocked(p.x,p.y,p.z)&&g.sim.collision.visible({...p,y:p.y+1.68},{...it.position,y:it.position.y+1.2}))view={p,yaw,length};}if(!view)throw Error('No accessible approach '+id);g.sim.state.player.position=view.p;g.sim.state.player.yaw=view.yaw;g.sim.state.player.pitch=Math.atan2(1.68-(id==='ranger-shelter'?.3:1.2),view.length);g.sim.state.player.stance='stand';g.sim.verticalVelocity=0;g.sim.grounded=true;g.sim.locomotion.reset();await g.renderer.world.stream(true);for(const a of Object.values(g.sim.state.actors))if(a.health>0&&Math.hypot(a.position.x-it.position.x,a.position.z-it.position.z)<25){a.health=0;a.deathTime=g.sim.state.elapsed;}},id);
await page.waitForTimeout(450);const target=await page.evaluate(()=>window.__game.renderer.target());if(!target)throw Error('no target '+id);
await page.keyboard.press('KeyE');if(target.id.startsWith('conversation:'))await page.locator('[data-action="narrative-interact"][data-id="narrative:'+id+'"]').click();else await page.waitForFunction(()=>!window.__game.sim.actions.pending);await page.waitForTimeout(100);
const status=await page.evaluate(()=>({quests:window.__game.sim.state.narrative.quests,frame:window.__game.sim.narrative.frame().id,error:window.__game.inspect().error}));if(status.frame){await page.keyboard.press('KeyX');await page.waitForTimeout(100);}results.push({id,target:target.id,...status});
}
await page.evaluate(async()=>{const g=window.__game,{SIDE_QUESTS}=await import('/src/narrative/quests.ts');const costs={fuel:1};for(const q of SIDE_QUESTS)for(const step of q.steps.slice(g.sim.state.narrative.quests[q.id]?.stage??0))for(const[id,n]of Object.entries(step.cost??{}))costs[id]=(costs[id]??0)+n;for(const[id,n]of Object.entries(costs)){const have=g.sim.state.player.inventory.items.filter(i=>i.id===id).reduce((a,i)=>a+i.count,0);if(n>have)g.command('give '+id+' '+(n-have));}});
for(const id of ['doctor-request','doctor-found','doctor-return','hunter-request','hunter-route','hunter-return','checkpoint-request','checkpoint-proof','checkpoint-return','raider-request','raider-deal','raider-return','radio-request','radio-repair','radio-return','shelter-request','shelter-power','shelter-return','officer-request','officer-proof','officer-return','mayor-record'])await step(id);
const completed=await page.evaluate(()=>window.__game.sim.narrative.sideQuests.map(q=>({id:q.id,completed:q.completed})));if(completed.some(q=>!q.completed))throw Error('incomplete quests '+JSON.stringify(completed));
await step('facility-enter');await page.evaluate(()=>{const g=window.__game;const it=g.sim.narrative.interactions().find(i=>i.id==='narrative:facility-control');g.sim.state.player.position={x:it.position.x,y:it.position.y,z:it.position.z-1.6};g.sim.state.player.yaw=0;g.sim.state.player.pitch=.1;});await page.waitForTimeout(250);
const base=await page.evaluate(()=>JSON.stringify(window.__game.sim.state));const endings=[];
for(const choice of ['publish','destroy','shutdown']){
await page.evaluate(async({base,choice})=>{const g=window.__game;const state=JSON.parse(base);state.name='QA 结局 '+choice;await g.game.setWorld(state);g.sim.god=true;g.game.input.fallback=true;g.ui.setLocked(true);g.ui.entityId='finale';g.game.open('conversation');}, {base,choice});await page.locator('[data-action="narrative-choice"][data-id="'+choice+'"]').click();await page.waitForTimeout(1800);await page.screenshot({path:'docs/phase2/screenshots/ending-'+choice+'-cinematic.png',scale:'css'});await page.keyboard.press('KeyX');await page.getByRole('button',{name:'留在灰谷，继续生存'}).waitFor();await page.screenshot({path:'docs/phase2/screenshots/ending-'+choice+'.png',scale:'css'});const ending=await page.evaluate(()=>({choice:window.__game.sim.state.narrative.choice,ending:window.__game.sim.state.narrative.ending,ended:window.__game.sim.state.ended,audio:window.__game.game.audio.diagnostics()}));await page.getByRole('button',{name:'留在灰谷，继续生存'}).click();await page.waitForTimeout(300);ending.continued=await page.evaluate(()=>!window.__game.sim.state.ended&&window.__game.ui.screen==='play');endings.push(ending);await page.keyboard.press('Escape');
}
return{setup:'Uses saved mainline fixture, developer travel/invulnerability and supplied quest costs. Quest and choice progression use actual E and UI; no completion flags injected.',results,completed,endings,errors};
}

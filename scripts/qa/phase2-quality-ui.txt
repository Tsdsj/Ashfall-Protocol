async(page)=>{
 const results=[],errors=[];page.on('pageerror',e=>errors.push(String(e)));
 await page.evaluate(()=>window.__game.game.open('settings'));
 for(const quality of ['low','medium','high','ultra']){
  await page.evaluate(()=>{window.__oldRenderer=window.__game.renderer;});await page.locator('#quality').selectOption(quality);await page.waitForFunction(()=>window.__game.renderer!==window.__oldRenderer&&!window.__game.game.loading&&window.__game.ui.screen==='settings',null,{timeout:60000});
  results.push(await page.evaluate(()=>{const g=window.__game,r=g.renderer;return {quality:g.game.settings.quality,ssao:!!r.lighting.ssao,shadow:r.lighting.shadows.getShadowMap().getSize().width,flashlightShadow:r.lighting.flashlightShadows.getShadowMap().getSize().width,density:r.world.density,resolution:r.stats.resolution,error:g.inspect().error}}));
 }
 await page.locator('[data-action="reset-settings"]').click();await page.waitForFunction(()=>!window.__game.game.loading&&window.__game.game.settings.quality==='high'&&window.__game.ui.screen==='settings',null,{timeout:60000});const reset=await page.evaluate(()=>({ssao:!!window.__game.renderer.lighting.ssao,quality:window.__game.game.settings.quality}));
 for(const size of [{width:920,height:680},{width:1280,height:800},{width:1600,height:900}]){
  await page.setViewportSize(size);await page.evaluate(()=>window.__game.game.open('inventory'));await page.waitForTimeout(250);const layout=await page.evaluate(()=>{const p=document.querySelector('.panel-body');return{width:innerWidth,documentWidth:document.documentElement.scrollWidth,panelWidth:p.clientWidth,panelScroll:p.scrollWidth}});if(layout.documentWidth>size.width+1||layout.panelScroll>layout.panelWidth+2)throw Error('horizontal overflow '+JSON.stringify(layout));results.push(layout);await page.screenshot({path:'docs/phase2/screenshots/inventory-'+size.width+'.png',scale:'css'});
 }
 await page.setViewportSize({width:1200,height:890});await page.evaluate(()=>window.__game.game.open('settings'));await page.locator('#setting-reducedMotion').check();await page.locator('#setting-subtitles').uncheck();await page.evaluate(()=>{const g=window.__game;g.sim.narrative.replayLog(g.sim.narrative.audioLogs[0]?.id??'opening-impact');g.ui.show('play');g.game.input.fallback=true;g.ui.setLocked(true);});await page.waitForTimeout(250);const subtitles=await page.locator('#sequence-text').isVisible()?await page.locator('#sequence-text').textContent():'';if(subtitles)throw Error('subtitles switch ignored');await page.keyboard.press('Escape');await page.evaluate(()=>window.__game.game.open('settings'));await page.locator('#setting-reducedMotion').uncheck();await page.locator('#setting-subtitles').check();await page.evaluate(()=>window.__game.game.open('pause'));return{results,reset,subtitlesDisabled:true,errors};
}
